Formal readings of AtomicStructure

12/-!
13# Formal readings of AtomicStructure
14
15Hydrogen energies and transition frequencies reuse PhysLib. Their identification
16with the Hamiltonian's point spectrum is explicitly not proved in PhysLib.
17The local Bohr model adds radius and angular-momentum formulas; it does not
18construct quantum eigenstates or solve classical electron orbits.
19
20Charge distributions, apparatus conditions, Rutherford deflection, and mercury
21excitation are local models. No hydrogen theorem is used to assert mercury
22excitation or cathode-ray measurements. Thresholds and tolerances are explicit
23inputs in consistent units, not estimates of historical experimental uncertainty.
24
25`DiscreteSpectralLines` means frequencies drawn from a countably indexed
26transition family, not a topologically discrete set: spectral series accumulate.
27The graph's existing edges are unchanged and are not certified by this module.
28See `AtomicStructure.md` for the node-by-node correspondence.
29-/

49/-- Material-independent negative electron charge and common subatomic mass.
50`Material` is the specified class of emitting materials, not proof of occurrence
51in every material in nature. -/
52def SubatomicElectronTheory (d : CathodeData Material) (electronMass : ℝ) : Prop :=
53 Nonempty Material ∧ 0 < electronMass ∧
54 ∀ material, d.charge material = -elementaryCharge ∧
55 d.mass material = electronMass ∧ electronMass < d.atomicMass material ∧
56 0 < d.emissionRate material
57
58/-- Magnitude of the specific charge, with an explicit reference value and
59dimensionless lower-bound factor making "high" relative rather than arbitrary. -/
60def UniversalChargeToMassRatio (d : CathodeData Material)
61 (referenceRatio highFactor : ℝ) : Prop :=
62 Nonempty Material ∧ 0 < referenceRatio ∧ 1 < highFactor ∧
63 (∀ material, 0 < d.mass material ∧ d.charge material < 0 ∧
64 highFactor * referenceRatio ≤ |d.charge material| / d.mass material) ∧
65 ∀ a b, |d.charge a| / d.mass a = |d.charge b| / d.mass b
66
67theorem electron_specific_charge (d : CathodeData Material) (m : ℝ)
68 (h : SubatomicElectronTheory d m) (a : Material) :
69 |d.charge a| / d.mass a = elementaryCharge / m := by
70 rw [(h.2.2 a).1, (h.2.2 a).2.1, abs_neg, abs_of_pos elementaryCharge_pos]
71
72theorem electron_ratio_universal (d : CathodeData Material) (m : ℝ)
73 (h : SubatomicElectronTheory d m) (reference factor : ℝ)
74 (hr : 0 < reference) (hf : 1 < factor)
75 (hhigh : factor * reference ≤ elementaryCharge / m) :
76 UniversalChargeToMassRatio d reference factor := by
77 refine ⟨h.1, hr, hf, ?_, ?_⟩
78 · intro a
79 refine ⟨?_, ?_, ?_⟩
80 · rw [(h.2.2 a).2.1]; exact h.2.1
81 · rw [(h.2.2 a).1]; exact neg_neg_of_pos elementaryCharge_pos
82 · rw [electron_specific_charge d m h a]; exact hhigh
83 · intro a b
84 rw [electron_specific_charge d m h a, electron_specific_charge d m h b]

98/-- A neutral ideal plum-pudding atom, including the normalization of the
99positive charge density. Integrals use ordinary Euclidean volume. -/
100def ThomsonModel (a : AtomData) : Prop :=
101 0 < a.radius ∧ 0 < a.positiveCharge ∧
102 (∀ x, a.positiveDensity x = if ‖x‖ ≤ a.radius then
103 a.positiveCharge / ((4 / 3 : ℝ) * π * a.radius^3) else 0) ∧
104 Integrable a.positiveDensity ∧ (∫ x, a.positiveDensity x) = a.positiveCharge ∧
105 a.electrons ≠ [] ∧
106 (∀ e ∈ a.electrons, ‖e.position‖ ≤ a.radius ∧ e.charge = -elementaryCharge) ∧
107 (a.electrons.map (·.charge)).sum = -a.positiveCharge
108
109/-- Charge and nearly all mass lie within a small central ball. Size and
110concentration fractions are explicit parameters, with values strictly below one. -/
111def RutherfordModel (a : AtomData) (nuclearRadius sizeRatio missingMassFraction : ℝ) : Prop :=
112 0 < a.radius ∧ 0 < a.positiveCharge ∧ 0 < a.totalMass ∧
113 0 < nuclearRadius ∧ 0 < sizeRatio ∧ sizeRatio < 1 ∧
114 nuclearRadius ≤ sizeRatio * a.radius ∧
115 0 ≤ missingMassFraction ∧ missingMassFraction < 1 ∧
116 Integrable a.positiveDensity ∧ Integrable a.massDensity ∧
117 (∀ x, 0 ≤ a.positiveDensity x ∧ 0 ≤ a.massDensity x) ∧
118 (∫ x, a.positiveDensity x) = a.positiveCharge ∧
119 (∫ x, a.massDensity x) = a.totalMass ∧
120 (∀ x, nuclearRadius < ‖x‖ → a.positiveDensity x = 0) ∧
121 (1 - missingMassFraction) * a.totalMass ≤
122 ∫ x in Metric.closedBall (0 : Vector) nuclearRadius, a.massDensity x

124/-- A stationary hydrogenic level scheme with the Bohr circular-radius and
125angular-momentum formulas. It is not a theorem about hydrogen eigenfunctions. -/
126def BohrModel (H : HydrogenAtom) (energy radius angularMomentum : ℕ → ℝ) : Prop :=
127 H.d = 3 ∧ 0 < H.k ∧ ∀ n,
128 energy n = H.energyLevel n ∧
129 radius n = (Constants.ℏ : ℝ)^2 * ((n : ℝ) + 1)^2 / (H.m * H.k) ∧
130 angularMomentum n = ((n : ℝ) + 1) * (Constants.ℏ : ℝ)

132theorem bohr_energy_ordered (H : HydrogenAtom) (E r L : ℕ → ℝ)

188def SmallAngleScatteringOnly (s : ScatteringData) (smallAngleBound : ℝ) : Prop :=
189 s.impactParameters.Nonempty ∧ 0 ≤ smallAngleBound ∧ smallAngleBound < π / 2 ∧
190 ∀ b ∈ s.impactParameters, 0 < b ∧ 0 ≤ s.deflection b ∧ s.deflection b ≤ smallAngleBound
191
192/-- Heavy-target, repulsive point-Coulomb kinematic model. The deflection formula
193is assumed, not derived from an ODE or from the nuclear density predicate. -/
194def RutherfordScatteringLaw (s : ScatteringData) (coupling energy : ℝ) : Prop :=
195 0 < coupling ∧ 0 < energy ∧
196 ∀ b ∈ s.impactParameters, 0 < b ∧
197 s.deflection b = 2 * arctan (coupling / (2 * energy * b))
198
199def LargeAngleBackscattering (s : ScatteringData) (coupling energy
200 targetMass projectileMass heavyTargetFactor : ℝ) : Prop :=
201 RutherfordScatteringLaw s coupling energy ∧
202 0 < projectileMass ∧ 1 < heavyTargetFactor ∧ heavyTargetFactor * projectileMass ≤ targetMass ∧
203 ∃ b ∈ s.impactParameters, π / 2 < s.deflection b ∧ s.deflection b < π
204
205theorem rutherford_large_deflection (s : ScatteringData) (k E b : ℝ)
206 (h : RutherfordScatteringLaw s k E) (hb : b ∈ s.impactParameters)
207 (hclose : 2 * E * b < k) : π / 2 < s.deflection b := by
208 have hpos : 0 < 2 * E * b := mul_pos (mul_pos (by norm_num) h.2.1) (h.2.2 b hb).1
209 have hratio : (1 : ℝ) < k / (2 * E * b) := (lt_div_iff₀ hpos).2 (by simpa using hclose)
210 have ha := Real.arctan_strictMono hratio
211 rw [Real.arctan_one] at ha
212 rw [(h.2.2 b hb).2]
213 linarith
214
215theorem rutherford_backscattering (s : ScatteringData) (k E M m factor b : ℝ)
216 (h : RutherfordScatteringLaw s k E) (hm : 0 < m) (hf : 1 < factor)
217 (hheavy : factor * m ≤ M) (hb : b ∈ s.impactParameters)
218 (hclose : 2 * E * b < k) : LargeAngleBackscattering s k E M m factor := by
219 refine ⟨h, hm, hf, hheavy, b, hb, rutherford_large_deflection s k E b h hb hclose, ?_⟩
220 rw [(h.2.2 b hb).2]
221 linarith [Real.arctan_lt_pi_div_two (k / (2 * E * b))]

223theorem small_large_incompatible (s : ScatteringData) (bound k E M m factor : ℝ)

230/-- A separate excitation-level and current-threshold model, usable for mercury.
231The current drop is a supplied one-sided-limit condition, not derived transport. -/
232def DiscreteEnergyTransfer (levels : ℕ → ℝ) (transfers : Set ℝ)
233 (current : ℝ → ℝ) (thresholds : Set ℝ) (dropBound : ℝ) : Prop :=
234 StrictMono levels ∧ transfers.Nonempty ∧
235 (∀ loss ∈ transfers, ∃ i j : ℕ, i < j ∧ loss = levels j - levels i) ∧
236 0 < dropBound ∧ thresholds.Nonempty ∧
237 (∀ loss ∈ transfers, ∃ V ∈ thresholds, elementaryCharge * V = loss) ∧
238 (∀ V ∈ thresholds, 0 < V ∧ ∃ loss ∈ transfers, elementaryCharge * V = loss) ∧
239 ∀ V ∈ thresholds, ∃ left right : ℝ,
240 Filter.Tendsto current (nhdsWithin V (Set.Iio V)) (nhds left) ∧
241 Filter.Tendsto current (nhdsWithin V (Set.Ioi V)) (nhds right) ∧ right + dropBound ≤ left

246def DiscreteSpectralLines (H : HydrogenAtom) (frequencies : Set ℝ) : Prop :=

249def frequenciesFromLevels (energy : ℕ → ℝ) : Set ℝ :=
250 {ν | ∃ lower upper : ℕ, lower < upper ∧

253theorem bohr_to_spectral_lines (H : HydrogenAtom) (E r L : ℕ → ℝ)

283/-- PhysLib uses zero-based level indices, so its lower index `1` is the
284physical principal quantum number `2` of the Balmer series. -/
285theorem balmer_formula (H : HydrogenAtom) (hd : H.d = 3) (upper : ℕ) :
286 H.balmerFrequency upper = H.rydbergFrequency *
287 (1 / 4 - 1 / ((upper : ℝ) + 1)^2) := by
288 have h := H.transitionFrequency_eq 1 upper
289 norm_num [HydrogenAtom.levelIndex, hd] at h
290 simpa only [one_div] using h
